315     * server is a 502, not a success with a quiet log note.
316     *

320    public function remote_action( string $action, WP_REST_Request $request ): WP_REST_Response|WP_Error {

322        $controller = $this->load_controller( $email );
323
324        if ( ! ( $controller instanceof Remote_Email_Controller_Interface ) ) {
325            return new WP_Error( 'bh_wp_mailboxes_not_remote', __( 'This email\'s account cannot act on the mail server.', 'bh-wp-mailboxes' ), array( 'status' => 400 ) );
326        }
327
328        try {
329            $controller = match ( $action ) {
330                'mark-read'        => $controller->mark_read_on_server(),
331                'mark-unread'      => $controller->mark_unread_on_server(),
332                'delete-on-server' => $controller->delete_on_server(),
333            };
334        } catch ( Invalid_Email_State_Exception $exception ) {
335            return new WP_Error( 'bh_wp_mailboxes_invalid_state', $exception->getMessage(), array( 'status' => 409 ) );
336        } catch ( Throwable $throwable ) {

338
339            return new WP_Error( 'bh_wp_mailboxes_remote_action_failed', $throwable->getMessage(), array( 'status' => 502 ) );
340        }
341
342        $email = $controller->get_email();
343
344        return new WP_REST_Response(
345            array(
346                'is_read'           => $email->is_remote_read,
347                'is_remote_deleted' => $email->is_remote_deleted,
348            )
349        );
350    }
