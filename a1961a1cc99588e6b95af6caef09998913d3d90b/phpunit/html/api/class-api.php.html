389        }

392
393        if ( is_null( $connection ) ) {
394            $this->logger->warning( 'No fetcher found for ' . $email_account->display_name );
395            return $failed(

397                    /* translators: %s: the account's connection class name */
398                    __( 'No connection is available for this account type (%s).', 'bh-wp-mailboxes' ),

401            );

408        }

410        if ( $connection instanceof Requires_Credentials ) {

414            if ( ! ( $credentials instanceof Account_Credentials_Interface ) ) {
415                $this->logger->warning( 'No credentials found for ' . $email_account->display_name );
416

423                    $this->logger->info(

425                        array( 'account_name' => $email_account->display_name )
426                    );
427
428                    return $skipped( __( 'Skipped: a login failed less than four hours ago. Check the credentials, or use "Check now" to try again immediately.', 'bh-wp-mailboxes' ) );
429                }
430            }

432        }

439                array(
440                    'exception'        => $exception,
441                    'account'          => $email_account->display_name,
442                    'mailbox_settings' => $email_account,
443                )
444            );

447
448            return $failed(
449                sprintf(
450                    /* translators: %s: the error message from the mail server or connection */
451                    __( 'Could not fetch emails: %s', 'bh-wp-mailboxes' ),
452                    $exception->getMessage()
453                )
454            );
455        }
456

458
459        // The fetch authenticated and completed, so record the successful login and check time.

461            $email_account,
462            last_checked_time: $now_time,
463            last_successful_login_time: $now_time,
464        );

466        $warnings = array();
467
468        $refresh_warning = $this->save_refreshed_credentials( $connection, $email_account );
469        if ( ! is_null( $refresh_warning ) ) {
470            $warnings[] = $refresh_warning;
471        }
472
473        // Drop any emails already saved locally (same account + Message-ID) so we never duplicate.
474        $all_new_account_emails = $all_new_account_emails->reject(
475            fn ( Fetched_Email $unsaved_email ): bool => $this->email_repository->is_post_for_message_id(
476                $email_account->email_address,
477                $unsaved_email->message->getMessageId() ?? ''
478            )

489
490        $new_emails_saved_count = count( $saved );
491
492        // Add this run to the account's lifetime totals (a historical record; they never decrease).
493        $total_emails_downloaded_count = $email_account->total_emails_downloaded_count + $new_emails_downloaded_count;
494        $total_emails_saved_count      = $email_account->total_emails_saved_count + $new_emails_saved_count;
495        if ( $new_emails_downloaded_count > 0 ) {
496            $this->email_account_repository->update(
497                $email_account,
498                total_emails_downloaded_count: $total_emails_downloaded_count,
499                total_emails_saved_count: $total_emails_saved_count,
500            );
501        }
502
503        $this->logger->debug(
504            sprintf(
505                '%s: %d emails retrieved since %s; %d already saved, %d new; %d filtered out, %d saved. Lifetime totals: %d downloaded, %d saved.',
506                $email_account->email_address,
507                $downloaded_count,
508                $since_datetime->format( DateTimeInterface::ATOM ),
509                $downloaded_count - $new_emails_downloaded_count,
510                $new_emails_downloaded_count,
511                $filtered_out_count,
512                $new_emails_saved_count,
513                $total_emails_downloaded_count,
514                $total_emails_saved_count
515            ),
516            array(
517                'account'                       => $email_account->email_address,
518                'since'                         => $since_datetime->format( DateTimeInterface::ATOM ),
519                'retrieved'                     => $downloaded_count,
520                'already_saved'                 => $downloaded_count - $new_emails_downloaded_count,
521                'new'                           => $new_emails_downloaded_count,
522                'filtered_out'                  => $filtered_out_count,
523                'saved'                         => $new_emails_saved_count,
524                'total_emails_downloaded_count' => $total_emails_downloaded_count,
525                'total_emails_saved_count'      => $total_emails_saved_count,
526            )
527        );
528
529        // If the mailbox is configured to mark-as-read or delete emails on the server after downloading,
530        // perform that action now. The mark/delete methods record their own log entry on each email.
531        $after_download_action = $email_account->after_download_remote_email_action();
532        if ( in_array( $after_download_action, array( 'mark_read', 'delete' ), true ) ) {
533            foreach ( $saved as $bh_email ) {
534                try {
535                    if ( 'mark_read' === $after_download_action ) {
536                        $this->mark_email_read( $bh_email );
537                    } else {
538                        $this->delete_email_on_server( $bh_email );
539                    }
540                } catch ( Throwable $exception ) {
541                    $this->logger->warning(
542                        'Post-download action "' . $after_download_action . '" failed: ' . $exception->getMessage(),
543                        array( 'post_id' => $bh_email->get_post_id() )
544                    );
545                    $warnings[] = sprintf(
546                        /* translators: 1: the action (mark_read or delete), 2: the email's post id, 3: the error message */
547                        __( 'Post-download action "%1$s" failed for email %2$d: %3$s', 'bh-wp-mailboxes' ),
548                        $after_download_action,
549                        $bh_email->get_post_id(),
550                        $exception->getMessage()
551                    );
552                }
553            }
554        }

557    }
558
559    /**
560     * Keep only the emails whose sender address matches the account's from-address regex and whose body
561     * matches its body-identifier regex.
562     *
563     * A null (or blank) regex matches everything. The body regex is tested against the plain-text body
564     * and, failing that, the HTML body. An invalid regex is reported as a warning and ignored, so a typo
565     * in a filter never silently discards mail.
566     *
567     * @see Email_Account_Settings_Interface::get_from_email_regex()
568     * @see Email_Account_Settings_Interface::get_body_identifier_regex()
569     *
570     * @param BH_Email_Account               $email_account The account whose filters to apply.
571     * @param Collection<int, Fetched_Email> $emails        The fetched emails not yet saved.
572     * @param string[]                       $warnings      Appended to with a message for each invalid regex.
573     *
574     * @return Collection<int, Fetched_Email> The emails that passed both filters, re-indexed.
575     */
576    protected function filter_by_account_regexes( BH_Email_Account $email_account, Collection $emails, array &$warnings ): Collection {
577
578        $from_regex = $this->validate_regex( $email_account->get_from_email_regex(), __( 'from address', 'bh-wp-mailboxes' ), $email_account, $warnings );
579        $body_regex = $this->validate_regex( $email_account->get_body_identifier_regex(), __( 'body identifier', 'bh-wp-mailboxes' ), $email_account, $warnings );
580
581        if ( is_null( $from_regex ) && is_null( $body_regex ) ) {
582            return $emails;
583        }
584
585        $filtered = $emails->filter(
586            function ( Fetched_Email $fetched ) use ( $from_regex, $body_regex, $email_account ): bool {
587                $message_id = $fetched->message->getMessageId() ?? '';
588                $context    = array(
589                    'account'    => $email_account->email_address,
590                    'message_id' => $message_id,
591                );
592
593                if ( ! is_null( $from_regex ) ) {
594                    $from_header = $fetched->message->getHeader( 'From' );
595                    $sender      = $from_header instanceof AddressHeader ? ( $from_header->getEmail() ?? '' ) : '';
596
597                    if ( 1 !== preg_match( $from_regex, $sender ) ) {
598                        $this->logger->debug( 'Not saving email ' . $message_id . ': sender "' . $sender . '" does not match the from address filter ' . $from_regex . '.', $context );
599                        return false;
600                    }
601                }
602
603                if ( ! is_null( $body_regex ) ) {
604                    $text = $fetched->message->getTextContent();
605                    $html = $fetched->message->getHtmlContent();
606
607                    $body_matches = ( ! is_null( $text ) && 1 === preg_match( $body_regex, $text ) )
608                        || ( ! is_null( $html ) && 1 === preg_match( $body_regex, $html ) );
609
610                    if ( ! $body_matches ) {
611                        $this->logger->debug( 'Not saving email ' . $message_id . ': body does not match the body identifier filter ' . $body_regex . '.', $context );
612                        return false;
613                    }
614                }
615
616                return true;
617            }
618        );
619
620        $dropped_count = $emails->count() - $filtered->count();
621        if ( $dropped_count > 0 ) {
622            $this->logger->info(
623                $dropped_count . ' of ' . $emails->count() . ' new emails did not match the account\'s filters and were not saved.',
624                array( 'account' => $email_account->email_address )
625            );
626        }
627
628        return $filtered->values();
629    }
630
631    /**
632     * Return the regex if it is usable, or null if it is unset, blank, or does not compile.
633     *
634     * @param ?string          $regex         The account setting.
635     * @param string           $label         Which filter this is, for the warning message.
636     * @param BH_Email_Account $email_account The account, for logging.
637     * @param string[]         $warnings      Appended to when the regex is invalid.
638     */
639    protected function validate_regex( ?string $regex, string $label, BH_Email_Account $email_account, array &$warnings ): ?string {
640        if ( is_null( $regex ) || '' === trim( $regex ) ) {
641            return null;
642        }
643
644        // preg_match() returns false and raises a warning (with PCRE's reason) for a pattern that does not compile.
645        $reason = '';
646        set_error_handler( // phpcs:ignore WordPress.PHP.DevelopmentFunctions.error_log_set_error_handler
647            function ( int $errno, string $errstr ) use ( &$reason ): bool {
648                $reason = $errstr;
649                return true;
650            }
651        );
652        try {
653            $compiles = false !== preg_match( $regex, '' );
654        } finally {
655            restore_error_handler();
656        }
657
658        if ( $compiles ) {
659            return $regex;
660        }
661
662        $message = sprintf(
663            /* translators: 1: which filter ("from address" or "body identifier"), 2: the regex as entered, 3: PCRE's error message */
664            __( 'The %1$s filter "%2$s" is not a valid regular expression and was ignored: %3$s', 'bh-wp-mailboxes' ),
665            $label,
666            $regex,
667            '' !== $reason ? $reason : preg_last_error_msg()
668        );
669
670        $this->logger->warning( $message, array( 'account' => $email_account->email_address ) );
671        $warnings[] = $message;
672
673        return null;
674    }

676    /**
677     * Validate an account's credentials by connecting to the server.
678     *
679     * If a connection doesn't need credentials, return the last email time and the user can use that to infer the health.
680     *
681     * @param BH_Email_Account               $account     The account whose connection to connect with.
682     * @param ?Account_Credentials_Interface $credentials Candidate credentials, or null to resolve via filter.
683     */
684    public function test_connection( BH_Email_Account $account, ?Account_Credentials_Interface $credentials = null ): Test_Connection_Result {
685
686        $connection = $this->get_connection_for_email_account( $account );
687
688        if ( is_null( $connection ) ) {
689            return new Test_Connection_Result( success: false, message: 'No email connection found for ' . $account->display_name . '.' );
690        }
691
692        if ( $connection instanceof Requires_Credentials ) {
693            $credentials = $credentials ?? $this->credentials_store->get( $account );
694
695            if ( ! ( $credentials instanceof Account_Credentials_Interface ) ) {
696                return new Test_Connection_Result( success: false, message: 'No credentials found for ' . $account->display_name . '.' );
697            }
698
699            $connection->set_credentials( $credentials );
700        }
701
702        try {
703            $connection->test_connection();
704
705            $this->save_refreshed_credentials( $connection, $account );
706
707            return new Test_Connection_Result( success: true, message: 'Connected successfully.' );
708        } catch ( Throwable $exception ) {
709            $this->logger->warning(
710                'Connection test failed for ' . $account->display_name . '. ' . $exception->getMessage(),
711                array(
712                    'account'   => $account->display_name,
713                    'exception' => $exception,
714                )
715            );
716
717            return new Test_Connection_Result( success: false, message: $exception->getMessage() );
718        }
719    }

721    /**
722     * Return the most recently downloaded emails.
723     *
